This satisfies the Check protocol from verdict/types.py.

Algorithm
Step 1 — Filter candidates. From added_functions, exclude:

is_test is True (tests are entry points, invoked by test runner)
Dunder names (__init__, __repr__, etc.) — called implicitly by Python
Decorator list contains pytest.fixture — injected by pytest
Decorator list contains app.route or starts with router. — framework handlers
Decorator list contains click.command or click.group — CLI entry points
Step 2 — Collect all referenced names in the repo. Walk every .py file under diff_root. Parse each with ast (skip on SyntaxError). Collect names from two node types only:

ast.Name → node.id (covers: func, func(), x = func, [func], @func)
ast.Attribute → node.attr (covers: module.func, obj.method())
Store as dict[str, set[tuple[str, int]]] mapping name → set of (file_path, line_number). This lets us distinguish where references come from. Walk the repo once, not once per candidate.

Step 3 — Self-recursion check. A function that only references itself inside its own body is still dead. When checking if a candidate is alive, verify at least one reference exists OUTSIDE the candidate's own line range (candidate line through end_line) in its own file.

Step 4 — Check __all__ and __init__.py. For each candidate, also check:

Is the name in an __all__ = [...] assignment in its own module file?
Is the name imported in any __init__.py file under diff_root?
If yes → alive, skip.
Step 5 — Emit findings. Each candidate with zero external references:

Finding(
    kind="dead_function",
    file=func["file"],
    line=func["line"],
    message=f"{func['name']}() is never referenced in the repo",
    confidence=0.9,
)
Confidence is 0.9 (not 1.0) because name-matching cannot catch dynamic references like getattr().

<content>"""Detect functions added in a diff that are never referenced in the repo."""

import ast
from pathlib import Path

from verdict.types import AddedFunction, Finding


class DeadFunctionCheck:
    """Check for added functions that are never referenced anywhere in the repo."""

    name = "dead_function"

    def run(
        self, diff_root: str, added_functions: list[AddedFunction]
    ) -> list[Finding]:
        """Find added functions with no references in the repo."""
        # Step 1: Filter candidates - exclude entry points
        candidates = [f for f in added_functions if not _is_excluded(f)]

        if not candidates:
            return []

        # Step 2: Collect all referenced names in the repo (walk once)
        references = _collect_references(diff_root)

        # Step 3-5: Check each candidate for external references
        findings: list[Finding] = []
        for func in candidates:
            func_name = func["name"]

            # Check if name is in __all__ or imported in __init__.py
            if _is_in_all_or_init(func_name, func["file"], diff_root):
                continue

            # Get all references to this name
            ref_locations = references.get(func_name, set())

            # Step 3: Self-recursion check - need at least one reference outside own body
            has_external_ref = False
            for ref_file, ref_line in ref_locations:
                # Reference in a different file -> alive
                if ref_file != func["file"]:
                    has_external_ref = True
                    break
                # Reference in same file but outside function body -> alive
                if ref_line < func["line"] or ref_line > func["end_line"]:
                    has_external_ref = True
                    break

            # Step 5: Emit finding if no external references
            if not has_external_ref:
                findings.append(
                    Finding(
                        kind="dead_function",
                        file=func["file"],
                        line=func["line"],
                        message=f"{func_name}() is never referenced in the repo",
                        confidence=0.9,
                    )
                )

        return findings


def _is_excluded(func: AddedFunction) -> bool:
    """Check if function should be excluded from dead code detection."""
    # Exclude test functions (invoked by test runner)
    if func["is_test"]:
        return True

    # Exclude dunder methods (called implicitly by Python)
    if func["name"].startswith("__") and func["name"].endswith("__"):
        return True

    # Check decorators for entry points
    for decorator in func["decorators"]:
        # Pytest fixtures
        if decorator == "pytest.fixture":
            return True
        # Flask/FastAPI route handlers
        if decorator == "app.route" or decorator.startswith("router."):
            return True
        # Click CLI commands
        if decorator in ("click.command", "click.group"):
            return True

    return False


def _collect_references(repo_root: str) -> dict[str, set[tuple[str, int]]]:
    """Walk all Python files and collect referenced names with locations."""
    references: dict[str, set[tuple[str, int]]] = {}
    repo_path = Path(repo_root)

    # Walk all .py files
    for py_file in repo_path.rglob("*.py"):
        try:
            source = py_file.read_text(encoding="utf-8")
            tree = ast.parse(source, filename=str(py_file))
        except (OSError, SyntaxError):
            # Skip files that can't be read or parsed
            continue

        # Get relative path for consistent comparison
        try:
            rel_path = str(py_file.relative_to(repo_path))
        except ValueError:
            # File is outside repo_root, use absolute path
            rel_path = str(py_file)

        # Walk AST and collect name references
        for node in ast.walk(tree):
            name: str | None = None
            line: int | None = None

            # ast.Name covers: func, func(), x = func, [func], @func
            if isinstance(node, ast.Name):
                name = node.id
                line = node.lineno
            # ast.Attribute covers: module.func, obj.method()
            elif isinstance(node, ast.Attribute):
                name = node.attr
                line = node.lineno

            if name and line:
                if name not in references:
                    references[name] = set()
                references[name].add((rel_path, line))

    return references


def _is_in_all_or_init(name: str, file_path: str, repo_root: str) -> bool:
    """Check if name is exported via __all__ or imported in __init__.py."""
    repo_path = Path(repo_root)
    file_full_path = repo_path / file_path

    # Check __all__ in the function's own module
    try:
        source = file_full_path.read_text(encoding="utf-8")
        tree = ast.parse(source, filename=str(file_full_path))

        for node in ast.walk(tree):
            # Look for __all__ = [...] assignments
            if isinstance(node, ast.Assign):
                for target in node.targets:
                    if isinstance(target, ast.Name) and target.id == "__all__":
                        # Check if our name is in the list
                        if isinstance(node.value, (ast.List, ast.Tuple)):
                            for elt in node.value.elts:
                                if isinstance(elt, ast.Constant) and elt.value == name:
                                    return True
    except (OSError, SyntaxError):
        pass

    # Check if imported in any __init__.py file
    for init_file in repo_path.rglob("__init__.py"):
        try:
            source = init_file.read_text(encoding="utf-8")
            tree = ast.parse(source, filename=str(init_file))

            for node in ast.walk(tree):
                # Check ImportFrom: from module import name
                if isinstance(node, ast.ImportFrom):
                    for alias in node.names:
                        if alias.name == name:
                            return True
                # Check Import: import module (less common for functions)
                elif isinstance(node, ast.Import):
                    for alias in node.names:
                        if alias.name == name:
                            return True
        except (OSError, SyntaxError):
            continue

    return False

+"""Detect functions added in a diff that are never referenced in the repo."""
+
+import ast
+from pathlib import Path
+
+from verdict.types import AddedFunction, Finding
+
+
+class DeadFunctionCheck:
+    """Check for added functions that are never referenced anywhere in the repo."""
+
+    name = "dead_function"
+
+    def run(
+        self, diff_root: str, added_functions: list[AddedFunction]
+    ) -> list[Finding]:
+        """Find added functions with no references in the repo."""
+        # Step 1: Filter candidates - exclude entry points
+        candidates = [f for f in added_functions if not _is_excluded(f)]
+
+        if not candidates:
+            return []
+
+        # Step 2: Collect all referenced names in the repo (walk once)
+        references = _collect_references(diff_root)
+
+        # Step 3-5: Check each candidate for external references
+        findings: list[Finding] = []
+        for func in candidates:
+            func_name = func["name"]
+
+            # Check if name is in __all__ or imported in __init__.py
+            if _is_in_all_or_init(func_name, func["file"], diff_root):
+                continue
+
+            # Get all references to this name
+            ref_locations = references.get(func_name, set())
+
+            # Step 3: Self-recursion check - need at least one reference outside own body
+            has_external_ref = False
+            for ref_file, ref_line in ref_locations:
+                # Reference in a different file -> alive
+                if ref_file != func["file"]:
+                    has_external_ref = True
+                    break
+                # Reference in same file but outside function body -> alive
+                if ref_line < func["line"] or ref_line > func["end_line"]:
+                    has_external_ref = True
+                    break
+
+            # Step 5: Emit finding if no external references
+            if not has_external_ref:
+                findings.append(
+                    Finding(
+                        kind="dead_function",
+                        file=func["file"],
+                        line=func["line"],
+                        message=f"{func_name}() is never referenced in the repo",
+                        confidence=0.9,
+                    )
+                )
+
+        return findings
+
+
+def _is_excluded(func: AddedFunction) -> bool:
+    """Check if function should be excluded from dead code detection."""
+    # Exclude test functions (invoked by test runner)
+    if func["is_test"]:
+        return True
+
+    # Exclude dunder methods (called implicitly by Python)
+    if func["name"].startswith("__") and func["name"].endswith("__"):
+        return True
+
+    # Check decorators for entry points
+    for decorator in func["decorators"]:
+        # Pytest fixtures
+        if decorator == "pytest.fixture":
+            return True
+        # Flask/FastAPI route handlers
+        if decorator == "app.route" or decorator.startswith("router."):
+            return True
+        # Click CLI commands
+        if decorator in ("click.command", "click.group"):
+            return True
+
+    return False
+
+
+def _collect_references(repo_root: str) -> dict[str, set[tuple[str, int]]]:
+    """Walk all Python files and collect referenced names with locations."""
+    references: dict[str, set[tuple[str, int]]] = {}
+    repo_path = Path(repo_root)
+
+    # Walk all .py files
+    for py_file in repo_path.rglob("*.py"):
+        try:
+            source = py_file.read_text(encoding="utf-8")
+            tree = ast.parse(source, filename=str(py_file))
+        except (OSError, SyntaxError):
+            # Skip files that can't be read or parsed
+            continue
+
+        # Get relative path for consistent comparison
+        try:
+            rel_path = str(py_file.relative_to(repo_path))
+        except ValueError:
+            # File is outside repo_root, use absolute path
+            rel_path = str(py_file)
+
+        # Walk AST and collect name references
+        for node in ast.walk(tree):
+            name: str | None = None
+            line: int | None = None
+
+            # ast.Name covers: func, func(), x = func, [func], @func
+            if isinstance(node, ast.Name):
+                name = node.id
+                line = node.lineno
+            # ast.Attribute covers: module.func, obj.method()
+            elif isinstance(node, ast.Attribute):
+                name = node.attr
+                line = node.lineno
+
+            if name and line:
+                if name not in references:
+                    references[name] = set()
+                references[name].add((rel_path, line))
+
+    return references
+
+
+def _is_in_all_or_init(name: str, file_path: str, repo_root: str) -> bool:
+    """Check if name is exported via __all__ or imported in __init__.py."""
+    repo_path = Path(repo_root)
+    file_full_path = repo_path / file_path
+
+    # Check __all__ in the function's own module
+    try:
+        source = file_full_path.read_text(encoding="utf-8")
+        tree = ast.parse(source, filename=str(file_full_path))
+
+        for node in ast.walk(tree):
+            # Look for __all__ = [...] assignments
+            if isinstance(node, ast.Assign):
+                for target in node.targets:
+                    if isinstance(target, ast.Name) and target.id == "__all__":
+                        # Check if our name is in the list
+                        if isinstance(node.value, (ast.List, ast.Tuple)):

+                                if isinstance(elt, ast.Constant) and elt.value == name:
+                                    return True

+    # Check if imported in any __init__.py file
+    for init_file in repo_path.rglob("__init__.py"):
+        try:
+            source = init_file.read_text(encoding="utf-8")
+            tree = ast.parse(source, filename=str(init_file))
+
+            for node in ast.walk(tree):
+                # Check ImportFrom: from module import name
+                if isinstance(node, ast.ImportFrom):
+                    for alias in node.names:
+                        if alias.name == name:
+                            return True
+                # Check Import: import module (less common for functions)
+                elif isinstance(node, ast.Import):
+                    for alias in node.names:
+                        if alias.name == name:
+                            return True
+        except (OSError, SyntaxError):

+
